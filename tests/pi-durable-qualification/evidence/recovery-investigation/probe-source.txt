function note(data) { const event = { at: new Date().toISOString(), ...data }; timeline.push(event); process.stdout.write(JSON.stringify(event) + '\n'); }
function docker(args) {
  const r = spawnSync('docker', ['--host=unix:///var/run/docker.sock', ...args], { env: environment, encoding: 'utf8', timeout: 30_000, maxBuffer: 16 * 1024 * 1024 });
  if (r.status !== 0) throw new Error(JSON.stringify({ args, status: r.status, error: r.error?.message, stderr: r.stderr }));
  return r.stdout;
}
async function request(c, route, body, conversation = 'restart') {
  const url = new URL(route, c.base);
  url.searchParams.set('conversation', conversation);
  const response = await fetch(url, { method: body ? 'POST' : 'GET', ...(body ? { body: JSON.stringify(body), headers: { 'content-type': 'application/json' } } : {}), signal: AbortSignal.timeout(20_000) });
  assert.equal(response.status, 200);
  const value = await response.json();
  note({ id: c.id, route, value });
  return value;
}

async function probe(c) {
  await request(c, '/submit', { requestId: 'run-A', text: 'first' });
  await request(c, '/submit', { requestId: 'run-B', text: 'second' });
  const pending = await request(c, '/pending');
  const before = await request(c, '/');
  assert.equal(pending.find(x => x.operationId === 'run-A').status, 'running');
  assert.equal(pending.find(x => x.operationId === 'run-B').status, 'queued');
  assert.ok(before.alarm);
  const killedAt = Date.now();
  assert.ok(Math.floor(before.alarm / 60_000) > Math.floor(killedAt / 60_000), 'Alarm minute must be after restart minute');
  docker(['kill', c.name]);
  docker(['start', c.name]);
  c.base = 'http://127.0.0.1:' + docker(['port', c.name, '9876']).trim().split(':').at(-1);
  note({ id: c.id, event: 'restarted', alarm: before.alarm, killedAt, defaultWakerTickMs: 60_000 });
  let wakeSnapshot;
  if (c.httpWake) { await delay(6000); wakeSnapshot = await request(c, '/snapshot'); }
  await delay(c.window);
  const preRequestLogs = docker(['logs', c.name]);
  await writeFile(c.project + '/logs-before-observation.txt', preRequestLogs);
  const snapshot = await request(c, '/snapshot');
  assert.equal(snapshot.run, undefined);
  assert.deepEqual(snapshot.inbox, []);
  const a = await request(c, '/wait?requestId=run-A');
  const b = await request(c, '/wait?requestId=run-B');
  assert.equal(a.status, 'done');
  assert.equal(b.status, 'done');
  const after = await request(c, '/');
  const result = { id: c.id, status: 'PASS', image: c.image, alarm: before.alarm, killedAt, observedAt: Date.now(), noHttpWindowMs: c.window, wakeSnapshot, snapshot, results: [a, b], beforeCounters: before.counters, afterCounters: after.counters, logsBeforeObservation: c.project + '/logs-before-observation.txt' };
  results.push(result);
  note({ id: c.id, status: result.status, elapsedMs: result.observedAt - killedAt });
}
